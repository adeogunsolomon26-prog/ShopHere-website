// 1. Sales Revenue by Staff Aggregation Pipeline: 
db.orders.aggregate([
  { $match: { orderStatus: "Received" } },
  { $lookup: { from: "employees", localField: "employeeId", foreignField: "_id", as: "employee" } },
  { $unwind: "$employee" },
  { $lookup: { from: "items", localField: "itemId", foreignField: "_id", as: "item" } },
  { $unwind: "$item" },
  { $group: {
      _id: { employeeId: "$employeeId", employeeName: { $concat: ["$employee.firstName", " ", "$employee.lastName"] }, month: { $month: "$orderDate" }, year: { $year: "$orderDate" } },
      totalOrders: { $sum: 1 },
      totalQuantity: { $sum: "$quantityReceived" },
      totalValue: { $sum: { $multiply: ["$quantityReceived", "$item.unitPrice"] } }
  }},
  { $sort: { "_id.year": -1, "_id.month": -1 } }
]);
// 2. Category Inventory Valuation Pipeline: 
db.items.aggregate([
  { $lookup: { from: "categories", localField: "categoryId", foreignField: "_id", as: "category" } },
  { $unwind: "$category" },
  { $group: {
      _id: "$category.categoryName",
      itemCount: { $sum: 1 },
      totalStock: { $sum: "$quantityInHand" },
      totalValuation: { $sum: { $multiply: ["$quantityInHand", "$unitPrice"] } }
  }},
  { $sort: { totalValuation: -1 } }
]);
